#+begin_src python
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
import numpy as np

matplotlib.rcParams['figure.dpi'] = 100

x = np.arange(5)
fig, ax = plt.subplots(figsize=(8, 4))
ax.bar(x, [30, 65, 45, 80, 55])
ax.set_title('Default DPI (100)')
ax.set_ylabel('Accuracy (%)')
print(f"DPI: {matplotlib.rcParams['figure.dpi']}")
plt.tight_layout()
plt.show()
#+end_src

* DPI 100 — 2x2 subplots (like our real plots)

#+begin_src python
matplotlib.rcParams['figure.dpi'] = 100

fig, axes = plt.subplots(2, 2, figsize=(14, 10))
for i, ax in enumerate(axes.flatten()):
    bars = np.random.randint(10, 90, size=8)
    x = np.arange(len(bars))
    ax.bar(x, bars, color=['#4878CF', '#6A5ACD', '#17BECF', '#E24A33', '#888', '#888', '#AAA', '#AAA'])
    ax.axhline(y=3.8, color='red', linestyle='--', alpha=0.5)
    ax.set_title(f'Encoder {i+1}', fontsize=12)
    ax.set_ylabel('Accuracy (%)')
    ax.set_xticks(x)
    ax.set_xticklabels(['Self\n(80/120)', 'Opus 4.5\n(45/120)', 'Sonnet\n(30/120)', 'Haiku\n(15/120)',
                         '', 'First\nletter\n(50/120)', 'Word\ncount\n(20/120)', 'Changed\nword\n(70/120)'], fontsize=6)
fig.suptitle('DPI 100 — 2x2 subplots', fontsize=13)
plt.tight_layout(rect=[0, 0, 1, 0.95])
plt.show()
#+end_src
